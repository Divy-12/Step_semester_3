public class Problem5Demo {

    public static class RaceEntry {
        private static int bibCounter = 0;
        private final String entryCode;
        private final String bibNumber;
        private final double entryFee;
        private double amountPaid;

        public RaceEntry(String bibNumber, double entryFee) {
            if (bibNumber == null || bibNumber.trim().length() < 4) {
                throw new IllegalArgumentException("Invalid bib number");
            }
            if (entryFee < 0) {
                throw new IllegalArgumentException("Entry fee cannot be negative");
            }

            bibCounter++;
            this.entryCode = "RUN-" + bibCounter;
            this.bibNumber = bibNumber.trim();
            this.entryFee = entryFee;
            this.amountPaid = 0.0;
        }

        // Single parameter overload
        public void pay(double amount) {
            if (amount > 0) {
                this.amountPaid += amount;
            }
        }

        // Two parameter overload reusing the single parameter version
        public void pay(double amount, String mode) {
            System.out.println("Paying via " + mode);
            pay(amount); // Reuses primary pay logic
        }

        public double getBalanceDue() {
            return Math.max(0.0, entryFee - amountPaid);
        }

        public String getEntryCode() { return entryCode; }
        public String getBibNumber() { return bibNumber; }
        public static int getBibCounter() { return bibCounter; }

        // Exact character-by-character validation without regex
        public static boolean isValidDiscountCode(String code) {
            if (code == null || code.length() != 5) {
                return false;
            }
            if (code.charAt(0) != 'M') {
                return false;
            }
            if (!Character.isDigit(code.charAt(1)) ||
                !Character.isDigit(code.charAt(2)) ||
                !Character.isDigit(code.charAt(3))) {
                return false;
            }
            return Character.isUpperCase(code.charAt(4));
        }
    }

    public static class RelayTeamEntry extends RaceEntry {
        private final int teamSize;

        public RelayTeamEntry(String bibNumber, double entryFee, int teamSize) {
            super(bibNumber, entryFee);
            this.teamSize = teamSize;
        }

        public int getTeamSize() { return teamSize; }
    }

    public static class EliteRunnerEntry extends RaceEntry {
        public EliteRunnerEntry(String bibNumber, double entryFee) {
            super(bibNumber, entryFee);
        }
    }

    // Settlement processor with null safety and instanceof separation
    public static String settleNight(RaceEntry[] entries) {
        int processed = 0;
        int nullSkipped = 0;
        int relayCount = 0;
        int individualCount = 0;

        if (entries != null) {
            for (RaceEntry entry : entries) {
                if (entry == null) {
                    nullSkipped++;
                    continue;
                }

                processed++;
                if (entry instanceof RelayTeamEntry) {
                    relayCount++;
                } else {
                    individualCount++;
                }
            }
        }

        return processed + " processed | " + nullSkipped + " null skipped | " +
               relayCount + " relay | " + individualCount + " individual";
    }

    public static void main(String[] args) {
        System.out.println("=== Problem 5 Tests ===");

        System.out.println("Discount Code Validation:");
        System.out.println("M123A: " + RaceEntry.isValidDiscountCode("M123A"));
        System.out.println("M12A:  " + RaceEntry.isValidDiscountCode("M12A"));
        System.out.println("X123A: " + RaceEntry.isValidDiscountCode("X123A"));

        System.out.println("\nPayment Overload Test:");
        EliteRunnerEntry eliteEntry = new EliteRunnerEntry("BIB3001", 150);
        eliteEntry.pay(10, "UPI");

        RelayTeamEntry relayEntry = new RelayTeamEntry("BIB4001", 300, 4);

        System.out.println("\nNightly Settlement:");
        RaceEntry[] batch = { eliteEntry, null, relayEntry };
        System.out.println(settleNight(batch));

        System.out.println("\nTotal Bib Counter: " + RaceEntry.getBibCounter());
    }
}
